VP2 <- function(xxx,yyy,zzz){ 
keyvals <- ifelse(xxx$padj > 0.05, 'gray', 
                  ifelse(xxx$hPTEC_LFC < 0, 'blue',
                         ifelse(xxx$hPTEC_LFC > 0, 'red', 'gray')))
keyvals[is.na(keyvals)] <- 'gray'
names(keyvals)[keyvals == 'red'] <- 'Up'
names(keyvals)[keyvals == 'gray'] <- 'Not changed'
names(keyvals)[keyvals == 'blue'] <- 'Down'
                EnhancedVolcano(xxx, lab = yyy, title="hPTEC", subtitle = "PBS vs CSRP3", x = 'hPTEC_LFC',  y = 'padj'  
                , selectLab = yyy %>% .[which(. %in% zzz)] 
                , boxedLabels = TRUE, drawConnectors = TRUE
                , pCutoff = 0.05,  cutoffLineType = "dotted"
                , pointSize = 1.5, labSize = 3, colCustom = keyvals, legendLabSize = 10, axisLabSize=8
                , colAlpha = 1, legendPosition = 'bottom', gridlines.major = FALSE, gridlines.minor = FALSE, caption=NULL)+
                coord_cartesian(xlim = c(-0.5, 0.5), ylim = c(0, 3))}
